-- Teachers jadvali
Create Table Teachers(
   TeacherId BigInt Identity(1, 1) Primary Key,
   FirstName Nvarchar(50) not null,
   LastName Nvarchar(50) not null,
   Age int not null,
   PhoneNumber Nvarchar(13) null,
   Grade Nvarchar(50) not null,
   Subject Nvarchar(50) null
);

-- Students jadvali
Create Table Students(
   StudentId BigInt Identity(1, 1) Primary Key,
   FirstName Nvarchar(50) not null,
   LastName Nvarchar(50) not null,
   Age int not null,
   PhoneNumber Nvarchar(13) null,
   Grade Nvarchar(50) not null
);

-- ClassRooms jadvali
Create Table ClassRooms(
    ClassRoomId BigInt Identity(1, 1) Primary Key,
    RoomNumber int Check(RoomNumber > 0 and RoomNumber < 100) not null
);

-- Teacher va Student o'rtasidagi ko'prik jadval
Create Table TeacherStudent(
    TeacherId BigInt,
    StudentId BigInt,
    Primary Key (TeacherId, StudentId),
    Constraint FK_TeacherStudent_Teacher Foreign Key(TeacherId) References Teachers(TeacherId),
    Constraint FK_TeacherStudent_Student Foreign Key(StudentId) References Students(StudentId)
);

-- ClassRoom va Teacher o'rtasidagi bog'lanish
Create Table ClassRoomTeachers(
    ClassRoomId BigInt,
    TeacherId BigInt,
    Primary Key (ClassRoomId, TeacherId),
    Constraint FK_ClassRoomTeachers_ClassRoom Foreign Key(ClassRoomId) References ClassRooms(ClassRoomId),
    Constraint FK_ClassRoomTeachers_Teacher Foreign Key(TeacherId) References Teachers(TeacherId)
);

-- ClassRoom va Student o'rtasidagi bog'lanish
Create Table ClassRoomStudents(
    ClassRoomId BigInt,
    StudentId BigInt,
    Primary Key (ClassRoomId, StudentId),
    Constraint FK_ClassRoomStudents_ClassRoom Foreign Key(ClassRoomId) References ClassRooms(ClassRoomId),
    Constraint FK_ClassRoomStudents_Student Foreign Key(StudentId) References Students(StudentId)
);
